                    \item \textbf{Limitation:} Lacks random read/write access; slow table scans for point lookups.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{\textbf{Review 2: Real-Time Operational NoSQL}}
                \begin{itemize}
                    \item \textbf{Framework:} \textbf{Apache HBase} on HDFS.
                    \item \textbf{Access Pattern:} Sub-second point lookups (\texttt{GET}), patient encounter range scans, and prefix filtering.
                    \item \textbf{Target:} Emergency triage, intensive care monitoring, 30-day readmission risk tracking.
                \end{itemize}
            \end{block}
        \end{column}

% Slide 3: Step 1 - Problem Statement & Motivation
\begin{frame}{Step 1: Application Introduction \& Problem Statement}
    \begin{itemize}
        \item \textbf{Domain from Syllabus:} \textit{Healthcare / Clinical Patient Record \& Encounter Management}.
        \item \textbf{The Core Challenges in Hospital Systems:}
        \begin{itemize}
            \item High write velocity from ICU vitals, emergency room intakes, and lab results.
            \item Traditional RDBMS exhibits write lock contention during hospital-wide surges.
            \item Auto-increment patient IDs cause \textbf{region hotspotting} in distributed databases.
        \end{itemize}
        \item \textbf{Why Apache HBase?}
        \begin{itemize}
            \item Distributed wide-column NoSQL database running directly on HDFS.
            \item Low-latency random read/write access with strong row-level consistency.
            \item Physical segregation of attributes into distinct column families stored in dedicated HFiles.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Step 3: Composite Row Key Engineering}
    \begin{center}
        \includegraphics[width=0.75\textwidth]{images/rowkey_design.png}
    \end{center}
    \vspace{-0.2cm}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \textbf{Anti-Hotspotting Architecture:}
            \begin{itemize}
                \item Sequential encounter IDs cause all admissions to hit a single RegionServer.
                \item Department salting (\texttt{Cardiology}, \texttt{Surgery}, \texttt{Pediatrics}) balances writes across nodes.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Clinical Query Benefits:}
            \begin{itemize}
                \item \textbf{Longitudinal Patient History:} Range scan directly on \texttt{Dept\#PatientNBR\#} pulls complete history.
                \item \textbf{Department Triage:} Prefix scans via \texttt{PrefixFilter('Cardiology\#')}.
                \item \textbf{Uniqueness:} Encounter ID suffix eliminates collisions.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

% Slide 9: Step 4 - Patient Range Scan & Filters (Part 1)
\begin{frame}[fragile]{Step 4: Longitudinal History Scan \& Clinical Filters}
    \begin{block}{\textbf{Patient Longitudinal Encounter History Range Scan}}
        \textit{Goal: Pull patient 55629189's complete encounter history in sub-5ms.}
        \begin{lstlisting}[language=bash]
scan 'patient_records', {STARTROW => 'Cardiology#55629189#', STOPROW => 'Cardiology#55629189~', LIMIT => 5}
        \end{lstlisting}
    \end{block}

    \begin{block}{\textbf{Filter 1: SingleColumnValueFilter -- High-Risk Readmission ($<$30 Days)}}
        \textit{Goal: Detect vulnerable patients readmitted within 30 days of discharge.}
        \begin{lstlisting}[language=bash]
scan 'patient_records', {FILTER => "SingleColumnValueFilter('patient', 'readmitted', =, 'binary:<30')", LIMIT => 5}
        \end{lstlisting}
    \end{block}

    \begin{block}{\textbf{Filter 2: SingleColumnValueFilter -- Polypharmacy Risk (num\_meds $>$ 15)}}
        \textit{Goal: Monitor patients on excessive concurrent medications to prevent drug interactions.}
        \begin{lstlisting}[language=bash]
scan 'patient_records', {FILTER => "SingleColumnValueFilter('clinical', 'num_meds', >, 'binary:15')", LIMIT => 5}
        \end{lstlisting}
    \end{block}
\end{frame}

% Slide 10: Step 4 - HBase Shell Filters (Part 2)
\begin{frame}[fragile]{Step 4: Advanced Analytical Filters (Continued)}
    \begin{block}{\textbf{Filter 3: PrefixFilter -- Department-Level Clinical Triage}}
        \textit{Goal: Retrieve all active patient admissions in the Cardiology Department.}
        \begin{lstlisting}[language=bash]
scan 'patient_records', {FILTER => "PrefixFilter('Cardiology#')", LIMIT => 5}
        \end{lstlisting}
    \end{block}

    \begin{block}{\textbf{Filter 4: RowFilter with RegexStringComparator -- Pediatric Encounters}}
        \textit{Goal: Track pediatric admissions across all medical centers.}
        \begin{lstlisting}[language=bash]
scan 'patient_records', {FILTER => "RowFilter(=, 'regexstring:^Pediatrics.*')", LIMIT => 5}
        \end{lstlisting}
    \end{block}

    \begin{block}{\textbf{Filter 5: ValueFilter with SubstringComparator -- Emergency Admissions}}
        \textit{Goal: Search across all columns for admissions originating from emergency rooms.}
        \begin{lstlisting}[language=bash]
scan 'patient_records', {FILTER => "ValueFilter(=, 'substring:EMERGENCY')", LIMIT => 5}
        \end{lstlisting}
    \end{block}
\end{frame}

% Slide 12: Step 5 - Java API Implementation
\begin{frame}[fragile]{Step 5: HBase Java API Implementation (Rubric: 2 Marks)}
    \begin{itemize}
        \item \textbf{Class:} \texttt{bigdata.HBaseHealthcareOperations.java}
        \item \textbf{Features:} Table Admin, Put, Get, Filter Scan, and Delete.
    \end{itemize}
    \begin{lstlisting}[language=Java]
Configuration config = HBaseConfiguration.create();
try (Connection conn = ConnectionFactory.createConnection(config);
     Table table = conn.getTable(TableName.valueOf("patient_records"))) {
    
    // Programmatic Filtered Scan in Java API
    Scan scan = new Scan();
    SingleColumnValueFilter filter = new SingleColumnValueFilter(
        Bytes.toBytes("patient"), Bytes.toBytes("readmitted"),
        CompareOperator.EQUAL, new BinaryComparator(Bytes.toBytes("<30"))
    );
    scan.setFilter(filter);
    scan.setLimit(5);

    try (ResultScanner scanner = table.getScanner(scan)) {
        for (Result r : scanner) {
            System.out.println("Match: " + Bytes.toString(r.getRow()));
        }
    }
}
    \end{lstlisting}
\end{frame}
