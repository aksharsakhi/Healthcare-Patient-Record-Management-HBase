# ------------------------------------------------------------------------------
# 6. APPLICATION-SPECIFIC ANALYTICAL QUERIES & FILTERS (RUBRIC: 4 MARKS)
# ------------------------------------------------------------------------------

# Filter 1: SingleColumnValueFilter - High-Risk 30-Day Hospital Readmissions ('<30')
scan 'patient_records', {FILTER => "SingleColumnValueFilter('patient', 'readmitted', =, 'binary:<30')", LIMIT => 5}

# Filter 2: SingleColumnValueFilter - Polypharmacy High-Risk Monitoring (num_medications > 15)
scan 'patient_records', {FILTER => "SingleColumnValueFilter('clinical', 'num_meds', >, 'binary:15')", LIMIT => 5}

# Filter 3: PrefixFilter - Rapid Department Triage (Cardiology Department)
scan 'patient_records', {FILTER => "PrefixFilter('Cardiology#')", LIMIT => 5}

# Filter 4: RowFilter with RegexStringComparator - Cross-Department Pediatric Encounters
scan 'patient_records', {FILTER => "RowFilter(=, 'regexstring:^Pediatrics.*')", LIMIT => 5}

# Filter 5: ValueFilter with SubstringComparator - Emergency Room Hospital Admissions
scan 'patient_records', {FILTER => "ValueFilter(=, 'substring:EMERGENCY')", LIMIT => 5}

# Filter 6: Compound Multi-Filter (FilterList MUST_PASS_ALL) - High-Priority: Cardiac Readmissions Within 30 Days
scan 'patient_records', {FILTER => "(PrefixFilter('Cardiology#')) AND (SingleColumnValueFilter('patient', 'readmitted', =, 'binary:<30'))", LIMIT => 5}

# Filter 7: ColumnPaginationFilter - Paginate Clinical Metrics for Mobile EHR Synching
scan 'patient_records', {FILTER => "ColumnPaginationFilter(3, 0)", LIMIT => 5}
